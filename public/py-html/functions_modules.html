def calculate_analytics(project_name, *metrics, **details):
    total = sum(metrics)
    avg = total / len(metrics) if metrics else 0
    print(f"Project: {project_name} | Avg: {avg:.1f}")
    for key, value in details.items():
        print(f"  - {key}: {value}")

calculate_analytics("Py-Learn-HTML", 95, 88, 92, author="Md Jaoyad SWE", status="Live")
