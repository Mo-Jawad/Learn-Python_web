1 Safe File Handling & Graceful Error Recovery

def load_python_project_config(filepath):
    try:
        print(f"[Attempting] Reading config from '{filepath}'...")
        with open(filepath, 'r') as file:
            return file.read()
    except FileNotFoundError:
        print("[Notice] File not found. Generating default system config...")
        return {"status": "healthy", "platform": "Py-Learn-HTML", "author": "Md Jaoyad SWE"}
    finally:
        print("[Clean-up] Resource handles successfully released.")

result = load_python_project_config("config.json")
print(f"Active Configuration: {result}")
